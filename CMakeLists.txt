cmake_minimum_required(VERSION 3.24)
project(NebulaCode VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)
include(GNUInstallDirs)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
endif()
add_library(nebula_workspace STATIC src/workspace/Workspace.cpp src/app/Commands.cpp src/app/Theme.cpp)
target_include_directories(nebula_workspace PUBLIC src)
target_link_libraries(nebula_workspace PUBLIC Qt6::Widgets)
if(BUILD_TESTING)
  add_executable(workspace_tests tests/workspace/WorkspaceTest.cpp resources/resources.qrc)
  target_link_libraries(workspace_tests PRIVATE nebula_workspace Qt6::Test)
  add_test(NAME workspace COMMAND workspace_tests -platform offscreen)
endif()
option(NEBULA_BUILD_APP "Build the desktop editor and its dependencies" ON)
if(NEBULA_BUILD_APP)
  include(cmake/Editor.cmake)
  include(cmake/Terminal.cmake)
  include(cmake/Services.cmake)
  add_library(nebula_workbench STATIC src/app/Workbench.cpp src/app/Sidebar.cpp src/app/ActivityBar.cpp src/app/AppearanceDialog.cpp)
  target_link_libraries(nebula_workbench PUBLIC nebula_workspace nebula_editor nebula_terminal nebula_services)
  add_executable(NebulaCode MACOSX_BUNDLE src/app/main.cpp src/platform/DesktopApplication.cpp resources/resources.qrc)
  target_link_libraries(NebulaCode PRIVATE nebula_workbench)
  if(APPLE)
    set(app_icon resources/icons/NebulaCode.icns)
    target_sources(NebulaCode PRIVATE "${app_icon}")
    set_source_files_properties("${app_icon}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(NebulaCode PROPERTIES MACOSX_BUNDLE_ICON_FILE NebulaCode.icns)
  endif()
  if(BUILD_TESTING)
    add_executable(workbench_tests tests/workspace/WorkbenchTest.cpp resources/resources.qrc)
    target_link_libraries(workbench_tests PRIVATE nebula_workbench Qt6::Test)
    add_test(NAME workbench COMMAND workbench_tests -platform offscreen)
    if(APPLE)
      add_executable(nebula_bench tests/performance/WorkbenchBench.cpp resources/resources.qrc)
      target_link_libraries(nebula_bench PRIVATE nebula_workbench Qt6::Test)
    endif()
  endif()
  set_target_properties(NebulaCode PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Nebula Code"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.nebulacode.editor" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
  install(TARGETS NebulaCode BUNDLE DESTINATION . RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    file(RELATIVE_PATH nebula_libdir
      "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_LIBDIR}")
    set_target_properties(NebulaCode PROPERTIES INSTALL_RPATH "\$ORIGIN/${nebula_libdir}")
    install(TARGETS qtermwidget6 LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}")
    install(FILES resources/nebula-code.desktop DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
    install(FILES resources/icons/NebulaCode.png
      DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps" RENAME nebula-code.png)
    install(FILES LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/nebula-code")
    install(FILES "${scintilla_SOURCE_DIR}/License.txt"
      DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/nebula-code" RENAME Scintilla.txt)
    install(FILES "${lexilla_SOURCE_DIR}/License.txt"
      DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/nebula-code" RENAME Lexilla.txt)
    install(FILES "${qtermwidget_SOURCE_DIR}/LICENSE.LGPL2+" "${qtermwidget_SOURCE_DIR}/LICENSE.BSD-3-clause"
      DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/nebula-code")
    set(CPACK_PACKAGE_NAME nebula-code)
    set(CPACK_PACKAGE_CONTACT "Nebula Code <https://github.com/saikrishna488/nebula-code/issues>")
    set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native Qt code editor")
    set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "qt6-qpa-plugins")
    set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "git, clangd")
    include(CPack)
  endif()
endif()
